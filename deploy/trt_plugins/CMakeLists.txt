cmake_minimum_required(VERSION 3.16)
project(agriocc_plugins LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64")
    set(CMAKE_CUDA_ARCHITECTURES 87)
  else()
    set(CMAKE_CUDA_ARCHITECTURES 89)
  endif()
endif()
if(CMAKE_VERSION VERSION_LESS "3.18")
  set(CMAKE_CUDA_FLAGS "${CMAKE_CUDA_FLAGS} -gencode arch=compute_${CMAKE_CUDA_ARCHITECTURES},code=sm_${CMAKE_CUDA_ARCHITECTURES}")
endif()
find_path(TRT_INCLUDE NvInfer.h HINTS ${TENSORRT_INCLUDE_DIR} /usr/include/aarch64-linux-gnu /usr/include/x86_64-linux-gnu)
find_library(TRT_LIBRARY NAMES nvinfer libnvinfer.so.10 libnvinfer.so.8 HINTS ${TENSORRT_LIBRARY_DIR} /usr/lib/aarch64-linux-gnu /usr/lib/x86_64-linux-gnu)
if(NOT TRT_INCLUDE OR NOT TRT_LIBRARY)
  message(FATAL_ERROR "TensorRT headers/library missing; set TENSORRT_INCLUDE_DIR/TENSORRT_LIBRARY_DIR")
endif()
add_library(agriocc_plugins SHARED plugin.cpp ../cuda_ops/msda.cu ../cuda_ops/preprocess.cu)
target_include_directories(agriocc_plugins PRIVATE ${TRT_INCLUDE} ${CMAKE_CUDA_TOOLKIT_INCLUDE_DIRECTORIES})
target_link_libraries(agriocc_plugins PRIVATE ${TRT_LIBRARY})
target_compile_options(agriocc_plugins PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3> $<$<COMPILE_LANGUAGE:CXX>:-O3>)
